      function validQuotationLine(l) {
        return l.keys().hasOnly(['description','quantityUnits','unitPriceCents','discountBps','rate','amountCents','discountCents','subtotalCents','taxCents','totalCents']) && shortText(l.description,500) &&
          l.quantityUnits is int && l.quantityUnits > 0 && l.quantityUnits <= 1000000000 && l.unitPriceCents is int && l.unitPriceCents >= 0 && l.unitPriceCents <= 1000000000 && l.quantityUnits * l.unitPriceCents <= 10000000000000 &&
          l.discountBps is int && l.discountBps >= 0 && l.discountBps <= 10000 && l.rate in [0,5,19] &&
          l.amountCents is int && l.amountCents == int((l.quantityUnits * l.unitPriceCents + 500) / 1000) &&
          l.discountCents is int && l.discountCents == int((l.amountCents * l.discountBps + 5000) / 10000) &&
          l.subtotalCents is int && l.subtotalCents == l.amountCents - l.discountCents &&
          l.taxCents is int && l.taxCents == int((l.subtotalCents * l.rate + 50) / 100) &&
          l.totalCents is int && l.totalCents == l.subtotalCents + l.taxCents;
      }
